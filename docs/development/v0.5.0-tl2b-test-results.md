~~~powershell
$root = 'C:\Users\hazimeyou\AppData\Local\Temp\tl2b-1314a9a4'
$env:YMM4_PATH = "$root\refs"
$env:YMM4DirPath = $env:YMM4_PATH
dotnet build .\YMMProjectManager\YMMProjectManager.csproj -c Release "-p:YMM4_PATH=$env:YMM4_PATH" --nologo
dotnet build .\YMMProjectManager.sln -c Release "-p:YMM4_PATH=$env:YMM4_PATH" --nologo
dotnet build .\YMMProjectManager.sln -c Debug "-p:YMM4_PATH=$env:YMM4_PATH" --nologo

$env:YMMPX_LIB_PATH = "$root\ymmpx\YmmpxLib.dll"
$env:TEMP = "$root\focused"

$env:YMM_TEST_FILTER = 'PreviewComparisonTests,SceneComparisonHandoffTests,SceneRenderCoordinatorTests'

$env:YMM_TEST_FILTER = ''

$env:TEMP = "$root\debug"

dotnet run --project .\YMMProjectManager.Tests -c Debug --no-build
~~~
